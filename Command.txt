# Test GET /coupons/{id}.
$API_ID = "3iredn6zwg"
$COUPON_ID = "319326de971100158f6fa0daeec978fba1f17e4b"
$COUPON_URL = "http://localhost:4566/_aws/execute-api/$API_ID/local/coupons/$COUPON_ID"
$getResponse = Invoke-RestMethod -Uri $COUPON_URL -Method GET
$getResponse | ConvertTo-Json -Depth 10 | Set-Content -Encoding UTF8 .\part2_get_by_id_output.json

# Test PUT /coupons/{id} and retrieve the saved update.
$updateBody = @{ title = "Updated Fairmont assessment offer"; campaign_id = "SPRING2027" } | ConvertTo-Json
$updateResponse = Invoke-RestMethod -Uri $COUPON_URL -Method PUT -ContentType "application/json" -Body $updateBody
$updateResponse | ConvertTo-Json -Depth 10 | Set-Content -Encoding UTF8 .\part2_update_output.json
$afterUpdate = Invoke-RestMethod -Uri $COUPON_URL -Method GET
$afterUpdate | ConvertTo-Json -Depth 10 | Set-Content -Encoding UTF8 .\part2_get_after_update_output.json

# Query the provider_id and campaign_id global secondary index.
@{ ":provider" = @{ S = "FAIRMONT" }; ":campaign" = @{ S = "SPRING2027" } } | ConvertTo-Json -Compress | Set-Content -Encoding ASCII .\part2_query_values.json
$gsiResult = aws --no-cli-pager --endpoint-url http://localhost:4566 --profile localstack dynamodb query --table-name coupons --index-name provider_id-campaign_id-index --key-condition-expression "provider_id = :provider AND campaign_id = :campaign" --expression-attribute-values file://part2_query_values.json | ConvertFrom-Json
$gsiResult | ConvertTo-Json -Depth 10 | Set-Content -Encoding UTF8 .\part2_gsi_query_output.json

# Save the table capacity and index configuration.
$tableResult = aws --no-cli-pager --endpoint-url http://localhost:4566 --profile localstack dynamodb describe-table --table-name coupons | ConvertFrom-Json
$tableResult | ConvertTo-Json -Depth 10 | Set-Content -Encoding UTF8 .\part2_table_output.json
